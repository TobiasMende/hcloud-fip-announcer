If the namespace is default-deny, the controller needs this egress:

apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: {{ include "fip.fullname" . }}
  namespace: {{ .Release.Namespace }}
spec:
  podSelector:
    matchLabels:
      {{- include "fip.selectorLabels" . | nindent 6 }}
  policyTypes: [Egress, Ingress]
  ingress:
    - ports:                      # Prometheus scrapes and kubelet probes
        - {port: {{ .Values.metrics.port }}, protocol: TCP}
        - {port: {{ .Values.healthProbePort }}, protocol: TCP}
  egress:
    - ports:                      # DNS
        - {port: 53, protocol: UDP}
        - {port: 53, protocol: TCP}
    - to:                         # Hetzner API (api.hetzner.cloud)
        - ipBlock: {cidr: 0.0.0.0/0}
      ports:
        - {port: 443, protocol: TCP}
    - to:                         # kube-apiserver: replace with its address
        - ipBlock: {cidr: <apiserver-ip>/32}
      ports:
        - {port: 6443, protocol: TCP}
    - to:                         # health checks against pod IPs: replace with
        - podSelector: {}         # the selector and ports of your member Services
          namespaceSelector: {}
      ports:
        - {port: 8080, protocol: TCP}
